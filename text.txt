export default function App() {

  const [tab, setTab] = useState('Table');

  const [favorite, setFavorie] = useState([]);

  const toggle = (id) => {
    setFavorie((prev) =>
      prev.includes(id)
        ? prev.filter((x) => x !== id)
        : [...prev, id]
    );
  };

  return (
    <SQLiteProvider
      databaseName={DATABASE_NAME}
      onInit={initDB}
    >

      <View style={styles.root}>

        <ExpoStatusBar style="light" />

        {/* Header */}
        <View style={[ styles.header, { paddingTop: topInset + 10 }]}>
          <Text style={styles.title}>{tab === 'Table' ? 'Table' : 'Kitchen'}</Text>
          <Pressable
            style={styles.settingsButton}
            onPress={() => {
              console.log('เปิดตั้งค่า');
            }}
          >
            <Text style={styles.settingsIcon}>⋮</Text>
          </Pressable>
        </View>

        {/* Body */}
        <View style={styles.body}>
          {tab === 'Table'
            ? (<TableScreen onToggle={toggle} favorite={favorite} />)
            : (<KitchenScreen onToggle={toggle} favorite={favorite} />)
          }
        </View>

        {/* Bottom Tab */}
        <View style={styles.tabbar}>
          <TabButton label="Table" onPress={() => setTab('Table')} active={tab === 'Table'}/>
          <TabButton label="Kitchen" onPress={() => setTab('Kitchen')} active={tab === 'Kitchen'}/>
        </View>
      </View>

    </SQLiteProvider>
  );
}

import { StatusBar as ExpoStatusBar } from 'expo-status-bar';
import { useState } from 'react';
import { View, Text, TouchableOpacity, Pressable } from 'react-native';
import { SQLiteProvider } from 'expo-sqlite';
import { styles } from "./src/styles/appStyles";
import { topInset } from "./src/styles/theme";
import TableScreen from "./src/screen/table/TableScreen";
import KitchenScreen from "./src/screen/kitchen/KitchenScreen";
import { DATABASE_NAME, initDB } from "./src/db/database";


export default function App() {

  const [tab, setTab] = useState('Table');

  return (
    <SQLiteProvider
      databaseName={DATABASE_NAME}
      onInit={initDB}
    >

      <View style={styles.root}>

        <ExpoStatusBar style="light" />

        {/* Header */}
        <View style={[ styles.header, { paddingTop: topInset + 10 }]}>
          <Text style={styles.title}>{tab === 'Table' ? 'Table' : 'Kitchen'}</Text>
          <Pressable
            style={styles.settingsButton}
            onPress={() => {
              console.log('เปิดตั้งค่า');
            }}
          >
            <Text style={styles.settingsIcon}>⋮</Text>
          </Pressable>
        </View>

        {/* Body */}
        <View style={styles.body}>
        {tab === 'Table' 
          ? ( <TableScreen /> ) 
          : ( <KitchenScreen /> )
        }
      </View>

        {/* Bottom Tab */}
        <View style={styles.tabbar}>
          <TabButton label="Table" onPress={() => setTab('Table')} active={tab === 'Table'}/>
          <TabButton label="Kitchen" onPress={() => setTab('Kitchen')} active={tab === 'Kitchen'}/>
        </View>
      </View>

    </SQLiteProvider>
  );
}


const TabButton = ({ label, onPress, active }) => {
  return (
    <TouchableOpacity
      style={styles.tab}
      onPress={onPress}
    >
      <Text style={[styles.tabText,active && styles.tabActive]}>
        {label}
      </Text>
    </TouchableOpacity>
  );
};
